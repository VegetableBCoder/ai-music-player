## 2.2 表定义（DDL）

```sql
-- 歌曲实体
CREATE TABLE song_entity (
  id                 INTEGER PRIMARY KEY AUTOINCREMENT,
  canonical_title    TEXT    NOT NULL,
  artists_key        TEXT    NOT NULL,      -- 排序去重后以 U+001F 连接
  display_artists    TEXT    NOT NULL,      -- 展示用全署名
  album_name         TEXT,
  album_artist       TEXT,
  album_release_date TEXT,
  cover_cache_path   TEXT,                  -- 懒提取的封面缓存文件
  play_count         INTEGER NOT NULL DEFAULT 0,
  complete_count     INTEGER NOT NULL DEFAULT 0,
  skip_count         INTEGER NOT NULL DEFAULT 0,
  interrupt_count    INTEGER NOT NULL DEFAULT 0,
  last_played_at     INTEGER,
  created_at         INTEGER NOT NULL
);
-- entityKey 精确相等 = 同一实体（I1 的实体侧保证）
CREATE UNIQUE INDEX idx_song_identity  ON song_entity(canonical_title, artists_key);
CREATE INDEX        idx_song_last_play ON song_entity(last_played_at DESC);
CREATE INDEX        idx_song_complete  ON song_entity(complete_count DESC);
CREATE INDEX        idx_song_album     ON song_entity(album_name, album_artist);

-- 实体 ↔ 演唱者（支撑歌手维度分组与计数）
CREATE TABLE song_artist (
  entity_id   INTEGER NOT NULL REFERENCES song_entity(id) ON DELETE CASCADE,
  artist_name TEXT    NOT NULL,
  position    INTEGER NOT NULL DEFAULT 0,
  PRIMARY KEY(entity_id, artist_name)
);
CREATE INDEX idx_song_artist_name ON song_artist(artist_name);

-- 音乐文件
CREATE TABLE music_file (
  id                INTEGER PRIMARY KEY AUTOINCREMENT,
  entity_id         INTEGER          REFERENCES song_entity(id) ON DELETE CASCADE, -- NULL = 尚未挂靠
  path              TEXT    NOT NULL,          -- 唯一性由唯一索引 index_music_file_path 保证
  file_name         TEXT    NOT NULL,
  size              INTEGER NOT NULL,
  format            TEXT    NOT NULL,
  duration_ms       INTEGER,
  is_representative INTEGER,                   -- 1 = 代表；NULL = 非代表（★ 见 §2.4 修正说明）
  analysis_status   TEXT    NOT NULL DEFAULT 'UNANALYZED',
  analysis_error    TEXT,
  error_kind        TEXT,
  added_at          INTEGER NOT NULL,
  analyzed_at       INTEGER
);
CREATE INDEX idx_file_entity ON music_file(entity_id);
CREATE INDEX idx_file_status ON music_file(analysis_status);
-- 原为内联 UNIQUE，Room 会生成为具名唯一索引
CREATE UNIQUE INDEX index_music_file_path ON music_file(path);
-- I2：每个实体至多一个代表文件（★ 见 §2.4 修正说明）
CREATE UNIQUE INDEX idx_file_representative
  ON music_file(entity_id, is_representative);

-- 分类 / 标签
CREATE TABLE category (
  id         INTEGER PRIMARY KEY AUTOINCREMENT,
  name       TEXT    NOT NULL UNIQUE,
  is_main    INTEGER NOT NULL DEFAULT 0,
  is_builtin INTEGER NOT NULL DEFAULT 0,
  sort_order INTEGER NOT NULL DEFAULT 0
);

CREATE TABLE tag (
  name        TEXT    PRIMARY KEY,           -- I5：名称全局唯一
  category_id INTEGER NOT NULL REFERENCES category(id),
  is_builtin  INTEGER NOT NULL DEFAULT 0,
  created_at  INTEGER NOT NULL
);
CREATE INDEX idx_tag_category ON tag(category_id);

CREATE TABLE entity_tag (
  entity_id  INTEGER NOT NULL REFERENCES song_entity(id) ON DELETE CASCADE,
  tag_id     TEXT    NOT NULL REFERENCES tag(name) ON DELETE CASCADE,
  created_at INTEGER NOT NULL,
  PRIMARY KEY(entity_id, tag_id)
);
CREATE INDEX idx_entity_tag_tag ON entity_tag(tag_id);

-- 歌词（I7：path 唯一 ⇒ 一个 .lrc 至多关联一个实体）
CREATE TABLE lyric (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  entity_id   INTEGER NOT NULL REFERENCES song_entity(id) ON DELETE CASCADE,
  path        TEXT    NOT NULL UNIQUE,
  source      TEXT    NOT NULL,   -- SAME_DIR | EXTERNAL_DIR | TAG_MATCH
  match_level INTEGER NOT NULL,   -- 1 | 2 | 3
  matched_at  INTEGER NOT NULL
);
CREATE INDEX idx_lyric_entity ON lyric(entity_id);

-- 播放队列（允许重复、有序）
CREATE TABLE queue_item (
  id         INTEGER PRIMARY KEY AUTOINCREMENT,
  position   INTEGER NOT NULL,
  entity_id  INTEGER NOT NULL REFERENCES song_entity(id) ON DELETE CASCADE,
  is_forced  INTEGER NOT NULL DEFAULT 0,
  created_at INTEGER NOT NULL
);
CREATE INDEX idx_queue_position ON queue_item(position);

-- 播放状态（单例，id 固定为 0）
CREATE TABLE playback_state (
  id                     INTEGER NOT NULL PRIMARY KEY,  -- 恒为 0（★ 见 §2.4 修正说明）
  current_queue_item_id  INTEGER REFERENCES queue_item(id) ON DELETE SET NULL,
  mode                   TEXT    NOT NULL DEFAULT 'LIST_LOOP',
  position_ms            INTEGER NOT NULL DEFAULT 0,
  session_max_position_ms INTEGER NOT NULL DEFAULT 0,
  session_started_at     INTEGER,
  updated_at             INTEGER NOT NULL
);

-- 实际播放顺序（随机模式「上一首」依据 + 最近播放）
CREATE TABLE play_history (
  id              INTEGER PRIMARY KEY AUTOINCREMENT,
  entity_id       INTEGER NOT NULL REFERENCES song_entity(id) ON DELETE CASCADE,
  play_session_id TEXT    NOT NULL,
  played_at       INTEGER NOT NULL
);
CREATE INDEX idx_history_played ON play_history(played_at DESC);

-- 分析批次
CREATE TABLE analysis_run (
  id            INTEGER PRIMARY KEY AUTOINCREMENT,
  status        TEXT    NOT NULL,          -- RUNNING | COMPLETED | ABORTED
  started_at    INTEGER NOT NULL,
  finished_at   INTEGER,
  new_count     INTEGER NOT NULL DEFAULT 0,
  skipped_count INTEGER NOT NULL DEFAULT 0,
  cleaned_count INTEGER NOT NULL DEFAULT 0,
  analyzed_ok   INTEGER NOT NULL DEFAULT 0,
  failed_count  INTEGER NOT NULL DEFAULT 0
);

-- 批次 ↔ 文件（「最近分析记录」的文件清单）
CREATE TABLE analysis_run_file (
  run_id  INTEGER NOT NULL REFERENCES analysis_run(id) ON DELETE CASCADE,
  file_id INTEGER NOT NULL REFERENCES music_file(id) ON DELETE CASCADE,
  PRIMARY KEY(run_id, file_id)
);

-- 扫描来源 / 外部歌词目录
CREATE TABLE scan_source (
  id         INTEGER PRIMARY KEY AUTOINCREMENT,
  kind       TEXT    NOT NULL,             -- MUSIC | LYRICS
  path       TEXT    NOT NULL,
  enabled    INTEGER NOT NULL DEFAULT 1,
  created_at INTEGER NOT NULL,
  UNIQUE(kind, path)
);

-- LLM 归一化结果缓存（内容寻址，与实体无外键关系）
CREATE TABLE llm_cache (
  cache_key   TEXT    PRIMARY KEY,   -- sha1(文件名 + 元数据指纹 + model + prompt 版本)
  result_json TEXT    NOT NULL,
  created_at  INTEGER NOT NULL
);
```

## 2.4 实现修正（Phase 2a 落地时确认）

Room 的注解体系表达不了下面两处 DDL，故做了等价替换。均已用真机 instrumented 测试
（`MusicDatabaseSchemaTest`，12 条）验证过实际生成的 schema。

| 位置 | 文档原样 | 落地 | 原因与等价性 |
| --- | --- | --- | --- |
| `music_file` 的 I2 兜底 | `is_representative INTEGER NOT NULL DEFAULT 0` + **部分**唯一索引 `WHERE is_representative = 1 AND entity_id IS NOT NULL` | `is_representative INTEGER`（可空）+ 复合唯一索引 `UNIQUE(entity_id, is_representative)` | `androidx.room.Index` 只有 `value/orders/name/unique` 四个字段，**不支持部分索引**。SQLite 的唯一索引中 `NULL` 互不冲突，因此「非代表文件」可以有任意多条，而每个实体的 `1` 至多一条 —— 保证强度等价。**注意：`is_representative` 不可写 `0`**，否则同实体的多个非代表文件会被判为冲突 |
| `playback_state` 的单例 | `id INTEGER PRIMARY KEY CHECK (id = 0)` | `id INTEGER NOT NULL PRIMARY KEY` | Room **不生成 `CHECK` 约束**（`CHECK` 只出现在它的 SQL 解析器里，用于迁移比对），而 SQLite 的 `ALTER TABLE` 不支持追加 `CHECK`（只能在 `CREATE TABLE` 时写），那张 `CREATE TABLE` 由 Room 生成 —— 无法表达。数据库层保底退化为**主键唯一性**；「只写 `id = 0`」由 `PlaybackStateDao` 保证（2b 用 DAO 测试覆盖） |

其余差异为无害的等价形式：内联 `UNIQUE` 由 Room 生成为具名唯一索引（`index_music_file_path` 等），
`Index.Order.DESC` 用于复现 `ORDER BY ... DESC` 的索引顺序。

# 4. 关键事务与不变量维护

> 以下方法全部标注 `@Transaction`，是数据层的唯一写入口；任何模块不得绕过。

**Phase 2b 落地时的三点澄清**（本节的措辞与现实有两处需要对齐）

1. **跨表事务用 `db.withTransaction` 而非 `@Dao` 的 `@Transaction`**：物理删除要调 `StorageSource`、
   会话事务要调多个 DAO，而 DAO 由 Room 实例化、注入不了协作者。`07 §4.6.2` 自己也是用
   `db.withTransaction` 组织跨 DAO 事务的。纯单表的多语句事务（挂靠、选举、标签 CRUD）仍用
   `@Dao abstract class` 上的 `@Transaction`。
2. **展示字段刷新（§4.3）只负责写入，取值由调用方传入**：它要以代表文件的**内嵌元数据**回填，
   而 `§1` 给 `:core:data` 的存储职责只有「物理删除文件与读取封面缓存路径」，**不含读元数据**。
   因此 `:core:data` 的 `refreshDisplayFields(entityId, albumName, albumArtist, albumReleaseDate)`
   收的是值，读元数据由持有 `MetadataReader` 的上层完成。§4.1 步骤 8、§4.4 的「刷新展示字段」
   同理 —— 它们是**流程**的一步，不是数据层事务内的一步。
3. **会话事务（`02 §5.4` 的 `recordSessionStart` / `settleSession`）不在 `:core:data`**：
   按 `07 §4.6.2`，它们由 `:core:playback` 的 `PlaybackStateRepositoryImpl` 用
   `db.withTransaction` 组织。数据层只提供它们依赖的原语：`bumpPlayCount`、`bumpSessionResult`、
   `PlaybackStateDao.putSession`、`PlayHistoryDao.addHistory/pruneHistory`。
   `play_count` 只在会话开始 +1 这条契约（I10）由这些原语的语义保证。

## 4.1 挂靠文件 `attachAnalysisResult(fileId, result, now)`

```
1. 若文件 status == LINKED → 直接返回（I4，幂等）
2. artistsKey = result.artists.map(::normalize).sorted().distinct().joinToString("\u001F")
3. entity = SongDao.findByIdentity(result.canonicalTitle, artistsKey)
4. entity == null → 插入 song_entity + song_artist 行（position 按归一化后顺序）
5. MusicFileDao.link(fileId, entity.id, now)
6. 代表文件选举（§4.2）
7. 标签写入：对每个 (category, tagName)
     - category 不在当前有效分类中 → 丢弃该条（AI 不得新建分类）
     - tag 不存在 → 插入 tag(category_id)
     - 插入 entity_tag（IGNORE 冲突）
8. 刷新展示字段（§4.3）
```

> `AnalysisOrchestrator` 侧的 `commitAnalysisSuccess` 只是本事务的调用包装（额外更新 `analysis_run` 计数），**不得复制挂靠逻辑**。

## 4.2 代表文件选举 `reelectRepresentative(entityId)`

```sql
-- 取体积最大者（并列时取 id 较小者，保证确定性）
SELECT id FROM music_file WHERE entity_id = :entityId ORDER BY size DESC, id ASC LIMIT 1;
-- 非代表必须置 NULL（不是 0）：UNIQUE(entity_id, is_representative) 在 NULL 上彼此不冲突，
-- 才能容下同一实体的多条非代表文件。写成 (id = :winner) 会产生 0，第二条非代表即撞唯一索引
-- —— 见 §2.4 的替代方案。
UPDATE music_file SET is_representative = CASE WHEN id = :winner THEN 1 ELSE NULL END
WHERE entity_id = :entityId;
```
- 由**唯一索引**（`UNIQUE(entity_id, is_representative)`）兜底：即使逻辑出错，也不可能出现两个代表文件。
- **硬性约束**：`is_representative` 只允许写 `1` 或 `NULL`，**永远不要写 `0`**。
- 没有文件时：不设置，实体进入「不可用」。

## 4.3 展示字段刷新 `refreshDisplayFields(entityId)`

以代表文件的内嵌元数据回填 `album_name` / `album_artist` / `album_release_date`；封面缓存路径作废（置空，由封面仓库下次访问时重建）。**I11** 由此保证。

## 4.4 删除序列

| 操作 | 事务内容 |
| --- | --- |
| **移除文件** `unlinkFile(fileId)` | ① 取 `entity_id`；② 删除该 `music_file` 行；③ 若实体已无文件 → 按「删除实体」处理；④ 否则重新选举代表文件 + 刷新展示字段 |
| **物理删除文件** `deleteFilePhysically(fileId)` | ① **先删磁盘**（`StorageSource.delete`），失败则中止并报错；② 事务内执行 `unlinkFile` 同款逻辑 |
| **删除实体** `deleteSong(entityId)` | ① 删除 `song_entity` 行（外键级联清掉 `music_file` / `song_artist` / `entity_tag` / `lyric` / `play_history` / `queue_item`）；② `playback_state.current_queue_item_id` 被 `SET NULL`；③ **不触碰磁盘**（I8） |

**顺序要求**：物理删除必须先删磁盘再改库——否则库已删、磁盘残留，用户失去清理入口。

## 4.5 标签 CRUD 与删除保护（I12）

```
addTag(categoryId, name):
  name 已存在 → 返回 ReuseExisting(existingCategoryId)（UI 引导复用）
  否则插入 tag → Added

deleteTag(name):
  songCountOfTag(name) > 0 → 拒绝 Conflict（"该标签下还有歌曲"）
  否则删除 → 级联清掉 entity_tag 的悬挂行

addCategory(name) → 插入 category（is_builtin = 0）

  否则删除

setMainCategory(id, isMain) → 更新 category.is_main（多选，不互斥）
```

# 5. Repository 实现要点

1. **不做分页**：V1 数据量为百~千级，列表统一返回 `Flow<List<...>>`，配合 `LazyColumn`；若未来超过万级再引入 Paging 3。
2. **避免 N+1**：歌曲列表的标签用「先取当页 entityIds，再批量查 `tagsOfSongs`」两步，在 Repository 内 `combine` 成 `SongListItem`。
3. **可播性判定**：`isPlayable` 由 `music_file` 中 `analysis_status = LINKED` 的记录数派生，**不落库**（避免脏字段）；列表查询用 `LEFT JOIN + COUNT` 聚合或子查询。
4. **Flow 组合**：`observeSongs` 由 `song_entity` 查询 + `tagsOfSongs` 查询 `combine`；排序通过 `SongSort` 映射到不同的 `@Query`（SQL 排序，不用内存排序）。
5. **写操作全部走 DAO 的事务方法**；Repository 只做参数校验与结果封装（`AddTagResult` / `DeleteTagResult` 等 sealed 类型）。

**排序映射**

| `SongSort` | SQL |
| --- | --- |
| `NAME` | `ORDER BY canonical_title COLLATE NOCASE` |
| `RECENT_PLAYED` | `ORDER BY last_played_at DESC NULLS LAST`（SQLite 用 `last_played_at IS NULL, last_played_at DESC`） |
| `COMPLETE_COUNT` | `ORDER BY complete_count DESC, canonical_title` |

> 不做拼音排序（需求已明确排除），中文按 Unicode 码位排序。

**筛选映射**

| 条件 | SQL |
| --- | --- |
| 按名称 | `canonical_title LIKE '%' \|\| :q \|\| '%'` |
| 按歌词（仅已关联） | `EXISTS (SELECT 1 FROM lyric l WHERE l.entity_id = s.id)` |
| 某歌手 | `EXISTS (SELECT 1 FROM song_artist a WHERE a.entity_id = s.id AND a.artist_name = :name)` |
| 某专辑 | `s.album_name = :album AND s.album_artist = :albumArtist` |
| 某标签 | `EXISTS (SELECT 1 FROM entity_tag et WHERE et.entity_id = s.id AND et.tag_id = :tag)` |

# 8. 迁移策略

- `exportSchema = true`，schema JSON 入库版本控制。
- **开发期**（未发布）允许 `fallbackToDestructiveMigration()`；**首个发布版本起必须移除**，改为显式 `Migration`。
- 迁移原则：只做加表 / 加列（SQLite `ALTER TABLE ADD COLUMN`），不重命名、不改类型；需要结构性变更时用「新表 + 数据搬运 + 换名」三段式。
- 每个迁移配套一个 `MigrationTest`（Room `MigrationTestHelper` + 上一版 schema JSON）。
