package com.aimusic.player.ui.component

import androidx.compose.runtime.Immutable
import com.aimusic.player.common.error.AppError

/**
 * 列表四态（`09 §3.3.1`、`§4.5.2`）。
 *
 * `Empty` 与 `EmptyFiltered` 分开，是因为两者的**动作不同**：库空 → 引导去扫描；
 * 有筛选但无命中 → 引导清除筛选（`09 §5.1`）。
 */
sealed interface ListUiState<out T> {
    data object Loading : ListUiState<Nothing>
    data class Content<T>(val items: List<T>) : ListUiState<T>
    data object Empty : ListUiState<Nothing>
    data object EmptyFiltered : ListUiState<Nothing>
    data class Error(val error: AppError) : ListUiState<Nothing>
}
```

package com.aimusic.player.ui.component

import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.compose.LocalLifecycleOwner
import androidx.lifecycle.repeatOnLifecycle

/**
 * 收集一次性事件（`09 §4.6`）。
 *
 * 用 `repeatOnLifecycle(STARTED)` 而不是直接 `LaunchedEffect { flow.collect() }`：
 * 退到后台时自动暂停收集，避免在不可见状态下弹 snackbar / 触发导航。
 */
@Composable
fun <T> CollectEvents(flow: Flow<T>, onEvent: (T) -> Unit) {
    val lifecycleOwner = LocalLifecycleOwner.current
    LaunchedEffect(flow, lifecycleOwner) {
        lifecycleOwner.lifecycle.repeatOnLifecycle(Lifecycle.State.STARTED) {
            flow.collect(onEvent)
        }
    }
}

package com.aimusic.player.ui.component

import androidx.compose.runtime.Immutable

/** 行在多选态的呈现（`09 §4.3.1`）。 */
@Immutable
sealed interface RowSelectionState {
    data object Hidden : RowSelectionState
    data class Visible(val selected: Boolean, val enabled: Boolean) : RowSelectionState
}

/**
 * 多选态状态模型（`09 §4.3.3`）。**无状态组件只读它，状态由 ViewModel 持有。**
 *
 * `selectableIds` 是「当前筛选结果中的可选集」——不可用歌曲（I3）与受保护项（I12）已被剔除，
 * `toggle` 与 `selectAll` 都以它为准，故这两类项天然不可勾选。
 *
 * **`T` 为何是泛型**：歌曲 / 文件 / 队列项用 `Long`，而某分类下的标签列表按 `09 §3.2.5`
 * 是勾选=标签名（`String`）。Kotlin 没有类型参数默认值，故调用处须显式写 `<Long>` / `<String>`。
 */
@Immutable
data class MultiSelectState<T>(
    val isActive: Boolean = false,
    val selectedIds: Set<T> = emptySet(),
    val selectableIds: Set<T> = emptySet(),
) {
    val selectedCount: Int get() = selectedIds.size
    val allSelected: Boolean get() = selectableIds.isNotEmpty() && selectedIds.containsAll(selectableIds)
    val canBatch: Boolean get() = selectedIds.isNotEmpty()

    fun toggle(id: T) = if (id in selectableIds) {
        copy(selectedIds = if (id in selectedIds) selectedIds - id else selectedIds + id)
    } else {
        this
    }

    fun selectAll() = copy(selectedIds = selectedIds + selectableIds)
    fun clearSelection() = copy(selectedIds = emptySet())

    fun rowStateFor(id: T): RowSelectionState =
        if (!isActive) {
            RowSelectionState.Hidden
        } else {
            RowSelectionState.Visible(selected = id in selectedIds, enabled = id in selectableIds)
        }
}

package com.aimusic.player.playback

import com.aimusic.player.data.db.MusicDatabase
import com.aimusic.player.playback.model.PlaybackUiState
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
import javax.inject.Singleton

/**
 * Phase 5 的**过渡实现**：不接 ExoPlayer，只把「入队过滤（I3）+ 当前曲目」落地，
 * 让歌曲库能编译、能真机浏览，且 `06 §8` 的三条入队过滤可测。
 *
 * **Phase 6 会替换为真实实现**（Media3 `ExoPlayer` + `MediaSessionService` + `QueueEngine`，
 * 见 `07 §4`）。届时本类删除；调用方只依赖 [PlaybackController] 接口，无需改动。
 *
 * **与计划初稿的差别（2026-09-18 按实现回写）**：构造器**不标 `@Inject`** —— `scope`
 * 要的是 `@ApplicationScope` 限定的那个，且 `now` 是测试接缝，二者都该由
 * `PlaybackModule` 的 `@Provides` 显式装配（`@Inject` 构造器拿不到限定符，Dagger 也不认默认值）。
 */
@Singleton
class NoopPlaybackController(
    private val db: MusicDatabase,
    private val scope: CoroutineScope,
    private val now: () -> Long = System::currentTimeMillis,
) : PlaybackController {

    private val _state = MutableStateFlow(PlaybackUiState())
    override val state: StateFlow<PlaybackUiState> = _state.asStateFlow()

    override fun playFromList(songs: List<Long>, startIndex: Int) {
        scope.launch {
            // I3：不可用歌曲不入队（07 §4.10 的入队过滤唯一入口）
            val playable = db.musicFileDao().playableIds(songs).toSet()
            // 不能用过滤后的下标取起点：songs 里可能混着不可用的歌，过滤会把下标前移，
            // 拿原下标去过滤后列表必然取错（甚至越界）。先按原下标定位到具体那首，再看是否可用。
            val startId = songs.getOrNull(startIndex)
            val current = startId?.takeIf { it in playable }
            _state.value = _state.value.copy(
                currentEntityId = current,
                isPlaying = current != null,
                queueSize = songs.count { it in playable },
            )
        }
    }

    override fun appendToQueue(songId: Long) {
        scope.launch {
            if (db.musicFileDao().playableIds(listOf(songId)).isEmpty()) return@launch

            db.queueDao().appendToQueue(songId, now())
            _state.value = _state.value.copy(queueSize = _state.value.queueSize + 1)
        }
    }

    override fun insertNext(songId: Long) {
        scope.launch {
            if (db.musicFileDao().playableIds(listOf(songId)).isEmpty()) return@launch

            // 当前位置由真实实现按播放状态给出；过渡实现没有播放中的项，故传 null（= 插到最前）
            db.queueDao().insertNext(songId, null, now())
            _state.value = _state.value.copy(queueSize = _state.value.queueSize + 1)
        }
    }
}

package com.aimusic.player.data.repository

import android.content.Context
import androidx.room.Room
import androidx.room.RoomDatabase
import androidx.sqlite.db.SupportSQLiteDatabase
import androidx.test.core.app.ApplicationProvider
import androidx.test.ext.junit.runners.AndroidJUnit4
import com.aimusic.player.data.db.MusicDatabase
import com.aimusic.player.data.insertFile
import com.aimusic.player.data.insertSong

import com.aimusic.player.data.model.SongScope
import com.aimusic.player.data.model.SongSort

import com.google.common.truth.Truth.assertThat
import kotlinx.coroutines.flow.first
import org.junit.After
import org.junit.Before
import org.junit.Test
import org.junit.runner.RunWith

/**
 * 防 N+1（`06 §8` 首行、`§4.1`）：列表装配必须**固定次数**查询，与歌曲数量无关。
 *
 * 断言「不随行数增长」而不是钉死某个次数：次数本身脆弱（改装配实现就变），
 * 而「1 首与 20 首的查询次数相等」是稳定的不变量。
 */
@RunWith(AndroidJUnit4::class)
class LibraryQueryCountTest {

    private lateinit var db: MusicDatabase
    private var queryCount = 0

    @Before
    fun setUp() {
        val context = ApplicationProvider.getApplicationContext<Context>()
        db = Room.inMemoryDatabaseBuilder(context, MusicDatabase::class.java)
            .allowMainThreadQueries()
            .addCallback(object : RoomDatabase.Callback() {
                override fun onOpen(db: SupportSQLiteDatabase) {
                    db.addQueryCallback { queryCount++ }
                }
            })
            .build()
    }

    @After
    fun tearDown() = db.close()

    @Test
    fun 装配查询次数不随歌曲数增长() = runDbTest {
        val repo = LibraryRepositoryImpl(db)

        insertLinkedSong(0)
        queryCount = 0
        repo.observeSongs(SongScope.Songs, SongSort.NAME).first()
        val forOne = queryCount

        for (i in 1 until 20) insertLinkedSong(i)
        queryCount = 0
        repo.observeSongs(SongScope.Songs, SongSort.NAME).first()
        val forTwenty = queryCount

        // 若退化成逐行查询，这里会变成 1 的 20 倍
        assertThat(forTwenty).isEqualTo(forOne)
    }

    /**
     * 一首「已挂靠且有 LINKED 文件」的歌。
     *
     * 必须有 LINKED 文件：可播性与标签都是按 id 批量查的，若一件文件都没有，
     * 装配会走 `ids.isEmpty()` 之外的空结果分支，测不到真正的装配路径。
     */
    private fun insertLinkedSong(index: Int) {
        val songId = db.insertSong(title = "歌$index", artistsKey = "歌手$index")
        val fileId = db.insertFile(path = "/m/song$index.mp3", size = 1024, status = "LINKED")
        db.linkFile(fileId, songId, isRepresentative = true)
    }
}
```
